/**
 * What table_groups_view() (R/table-groups-view.R) adds to the TableModel of
 * inst/js/table/table.d.ts for the groups layout. A text file for the JS
 * owner to paste into table.d.ts; nothing loads it.
 *
 * Added to the existing types:
 *
 *   TableCol.mark?: TableMark | null
 *   TableModel.fold?: string | null
 *
 * Positions and widths in the marks are percentages of the cell (0 to 100,
 * two decimals, rounded in R). A per-row array has one entry per row, null
 * where the cell draws nothing there. Text is plain; a tooltip is not markup.
 *
 * What goes where for the groups layout:
 *
 *   TableCol.disp   plain text of every row, always filled (search reads it).
 *                   A mark's value label, or what the label would say when
 *                   the Values switch hides it; a swimlane's span tooltips;
 *                   a colour split's "level label, level label". null where
 *                   the cell is empty. It is shown beside the mark only when
 *                   the mark has `dw`.
 *   TableCol.key    the sort key of the row (a number; none for text).
 *   TableCol.num    true for a plain number only. Marks are left-aligned and
 *                   sort on `key`.
 *   TableCol.keys   with the drill on, a facet column's level: the claim a
 *                   click on the cell adds to the row's claim,
 *                   {grp: ["g1"]}. The header cells over the column carry
 *                   the same claim; a facet spanner (by_level layout) claims
 *                   its level.
 *   TableRows.indent / strong
 *                   a two-level `by`: parent rows 0 and strong, child rows 1.
 *                   Parent rows carry their own numbers, so they are rows,
 *                   not sections. TableModel.collapsed starts them folded.
 *   TableRows.keys  with the drill on, the row's path: a parent row
 *                   {grp: ["g1"]}, a child row {grp: ["g1"], cat: ["a"]},
 *                   a row of a one-level `by` {cat: ["a"]}. A click on a
 *                   colour segment adds {[mark.level_col]: [level.name]}.
 *   TableModel.fold the "Other: 3 groups below the cut" note under the last
 *                   row, hidden while a search is active.
 */

/** A claim-free scale: what the positions of a mark are percentages of. */
interface TableMarkDomain {
  min: number;
  max: number;
  /** min and max are days since 1970-01-01 (a swimlane or sparkline over
   *  dates). */
  date?: boolean;
}

/** The column axis drawn in the header (and under a `large` swimlane):
 *  tick positions in the domain's units and their final labels. A percent
 *  scale carries its "%" on the last label. Absent without an axis, or with
 *  fewer than two ticks. Position of a tick in the strip:
 *  (at - domain.min) / (domain.max - domain.min) * 100. */
interface TableMarkTicks {
  at: number[];
  labels: string[];
}

/** The hover card's description of the column. */
interface TableMarkHover {
  head?: string;
  sub?: string;
  /** What the mark measures ("N", "Sum of x"). */
  measure?: string;
  /** The column a colour split names its levels by. */
  color_by?: string;
  /** The percentage base (a count's denominator). */
  base?: number;
  /** The statistic words of a distribution: center, range, whisk, from, to. */
  words?: Record<string, string>;
}

interface TableMarkBase {
  /** Width in ch of the value label's slot, one per column: set when the
   *  mark draws a label (TableCol.disp) beside itself. Also the pad at the
   *  end of the header axis. */
  dw?: number;
  /** The "(43%)" tail after a count, per row. Only with `dw`. */
  pct?: (string | null)[];
  domain?: TableMarkDomain;
  ticks?: TableMarkTicks | null;
  hover?: TableMarkHover;
}

/** A plain number or text keeps no mark. A `num` mark exists for the "(43%)"
 *  tail of a count, and for a failed custom summary. */
interface TableMarkNum extends TableMarkBase {
  kind: 'num';
  failed?: boolean;
}

/** A tile: a painted square with the value in it. null fill is no square. */
interface TableMarkTile extends TableMarkBase {
  kind: 'tile';
  fill: (string | null)[];
  ink: (string | null)[];
}

interface TableMarkBar extends TableMarkBase {
  kind: 'bar';
  /** Track fill as a share of the cell; null draws no fill. */
  width: (number | null)[];
  fill?: string | null;
  /** Child rows of a nested table draw the lighter track. Absent when no
   *  row is a child (equal to rows.indent > 0). */
  child?: boolean[];
}

/** A diverging bar: half the cell each way. */
interface TableMarkBarDiv extends TableMarkBase {
  kind: 'bardiv';
  /** 0 to 50, from the middle. */
  width: (number | null)[];
  positive: boolean[];
}

/** One level of a colour-split bar, box, point range or pair. */
interface TableMarkLevel {
  name: string;
  color: string;
  /** Bar levels: the segment's width, its value in the measure's unit (zero
   *  draws no segment), the number printed inside it (stacked and percent
   *  modes), the ink for that number, the value label (grouped mode). */
  width?: (number | null)[];
  amount?: (number | null)[];
  inside?: (string | null)[];
  ink?: string;
  label?: (string | null)[];
  /** Box, point range and pair levels also carry the glyph fields below. */
  [glyphField: string]: unknown;
}

/** Colour split bar. A click on a segment claims
 *  {[level_col]: [levels[j].name]} (only with the drill on). */
interface TableMarkBarSplit extends TableMarkBase {
  kind: 'barsplit';
  mode: 'stacked' | 'percent' | 'grouped';
  levels: TableMarkLevel[];
  level_col?: string;
}

/** The statistics behind a glyph, in the data's units. A key is present
 *  when any row has it. */
interface TableMarkStats {
  center?: (number | null)[];
  low?: (number | null)[];
  high?: (number | null)[];
  whisker_low?: (number | null)[];
  whisker_high?: (number | null)[];
}

/** Box: two whisker segments (never through the body), caps, the box, the
 *  median tick. Positions are where each piece starts, widths how far it
 *  runs; null draws nothing. With `levels` (a colour split) the geometry
 *  fields move into each level. */
interface TableMarkBox extends TableMarkBase {
  kind: 'box';
  /** No outer range to read against: the lane draws a hairline rail. */
  bare?: boolean;
  whisker_low?: (number | null)[];
  whisker_low_width?: (number | null)[];
  box_low?: (number | null)[];
  box_width?: (number | null)[];
  center?: (number | null)[];
  box_high?: (number | null)[];
  whisker_high_width?: (number | null)[];
  whisker_high?: (number | null)[];
  /** Rows behind the glyph. */
  n?: (number | null)[];
  /** Plain-text tooltip; null for an empty cell. */
  tip?: (string | null)[];
  stats?: TableMarkStats;
  levels?: TableMarkLevel[];
}

/** Point range (and the dot): the outer range as a fence band, the inner
 *  range as a bar on it, the centre as a dot. outer_* only when the column
 *  has an outer range. */
interface TableMarkPointRange extends TableMarkBase {
  kind: 'pointrange';
  bare?: boolean;
  outer_low?: (number | null)[];
  outer_width?: (number | null)[];
  center?: (number | null)[];
  inner_low?: (number | null)[];
  inner_width?: (number | null)[];
  n?: (number | null)[];
  tip?: (string | null)[];
  stats?: TableMarkStats;
  levels?: TableMarkLevel[];
}

/** Pair (dumbbell): band, reference line, the segment, then the two ends. */
interface TableMarkPair extends TableMarkBase {
  kind: 'pair';
  /** The reference line, once per column. */
  ref?: number;
  /** Band colour per row (single glyph; a level has `color`). */
  fill?: (string | null)[];
  from: (number | null)[];
  to: (number | null)[];
  link_left: (number | null)[];
  link_width: (number | null)[];
  band_left: (number | null)[];
  band_width: (number | null)[];
  /** The segment is dashed. */
  dash: boolean[];
  /** The `to` end is outside the band: drawn open. */
  open: boolean[];
  tip: (string | null)[];
  /** Level only: "30.5 -> 53 (+22.5)" without the words. */
  ends?: (string | null)[];
  levels?: TableMarkLevel[];
}

/** A span of a swimlane: [left, width, index into fills (from 0), label?]. */
type TableMarkSpan = [number, number, number] | [number, number, number, string];

/** Swimlane: spans per row, a tooltip per span. `large` also draws the axis
 *  under the table. */
interface TableMarkInterval extends TableMarkBase {
  kind: 'interval';
  spans: TableMarkSpan[][];
  tips: string[][];
  fills: string[];
  large?: boolean;
}

/** Sparkline: point strings "x,y x,y" in a 100 by 36 box (y grows down), the
 *  band polygon, the last point as a dot in percent, the points' values as
 *  comma-joined text for the hover readout, and the reference line and band
 *  as y coordinates in the same box (once per column). */
interface TableMarkSparkline extends TableMarkBase {
  kind: 'sparkline';
  line: string[];
  band: (string | null)[];
  end_x: (number | null)[];
  end_y: (number | null)[];
  x_text: string[];
  y_text: string[];
  n: number[];
  ref_line?: number;
  ref_band_top?: number;
  ref_band_height?: number;
}

type TableMark =
  | TableMarkNum | TableMarkTile | TableMarkBar | TableMarkBarDiv
  | TableMarkBarSplit | TableMarkBox | TableMarkPointRange | TableMarkPair
  | TableMarkInterval | TableMarkSparkline;
